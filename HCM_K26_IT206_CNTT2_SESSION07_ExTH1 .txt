def sum_list(head):
    """Tính tổng các giá trị trong Linked List."""
    total = 0                    # tên rõ nghĩa thay cho "s"
    current = head
    while current is not None:   # so sánh None đúng chuẩn Python
        if not isinstance(current.data, (int, float)):
            raise TypeError("Dữ liệu trong node phải là số")  # bắt lỗi dữ liệu sai
        total += current.data
        current = current.next
    return total
